\section{Proof Details}
\label{app:proofs}

Unless stated otherwise, all proofs use Assumption~\ref{ass:honest-separation}:
every nuisance, ratio, candidate function, and coefficient is fixed before the
independent RCT and OBS evaluation observations are used.

\subsection{Conditional validity of the trial pseudo-outcome}

Fix $q=(q_0,q_1)$ without using the evaluation samples and condition on
$X$. Under trial randomization, consistency, and honest sample separation,
\begin{align}
\mathbb E_R\left[\frac{A}{e(X)}\{Y-q_1(X)\}\mid X\right]
&=\mathbb E_R\{Y(1)-q_1(X)\mid X\},
\\
\mathbb E_R\left[\frac{1-A}{1-e(X)}\{Y-q_0(X)\}\mid X\right]
&=\mathbb E_R\{Y(0)-q_0(X)\mid X\}.
\end{align}
Substituting these two equalities into the definition of $\varphi$ cancels the
two fitted regressions and yields
$\mathbb E_R\{\varphi(V;q)\mid X\}=\tau(X)$.  Applying the identity to
$q=\widehat\mu_R$, $q=\widehat\mu_O$, and their affine blend gives
\begin{align}
\mathbb E_R(Z_\lambda\mid X)&=\tau(X),
&
\mathbb E_R(\Delta\mid X)&=0.
\end{align}

\subsection{Proof of Corollary~\ref{cor:rct-only-aipw}}

Apply Equation~\eqref{eq:pseudo-valid} with the nuisance-sample fitted pair
$q=\widehat\mu_R$, average over $D_R^{\mathrm{eval}}$, and use
iterated expectation. This gives
\begin{align*}
\mathbb E\{\widehat\theta_{\mathrm{AIPW},R}\}
&=\mathbb E_R\{\tau(X)\}=\theta.
\end{align*}

\subsection{Proof of Theorem~\ref{thm:ate-variance}}

Pseudo-outcome validity and exact transport give
\begin{align*}
P_R(Z_\lambda)&=\theta,
&
P_O(r_0\widehat{g})&=P_R(\widehat{g}).
\end{align*}
Substituting these identities into Equation~\eqref{eq:ate-estimator} proves
Equation~\eqref{eq:ate-exact-target}. For any fixed $\widetilde r$,
\begin{align*}
\mathbb E\{\widehat\theta_{\widetilde r}(\lambda,\omega)\}-\theta
&=\omega\{P_O(\widetilde r \widehat{g})-P_R(\widehat{g})\}
\\
&=\omega P_O\{(\widetilde r-r_0)\widehat{g}\},
\end{align*}
which proves Equation~\eqref{eq:ate-ratio-drift}. Using
the same pseudo-outcome and transport identities, rewrite the centered estimator as
\begin{align}
\widehat\theta_{r_0}(\lambda,\omega)-\theta
&=
(\mathbb P_{R,n}-P_R)\{Z_\lambda-\omega \widehat{g}\}
+(\mathbb P_{O,N}-P_O)\{\omega r_0\widehat{g}\}.
\end{align}
The two summands are independent because $D_R^{\mathrm{eval}}$ and
$D_O^{\mathrm{eval}}$ are
independent. The variance of
an empirical mean of $m$ independent observations is the population variance
divided by $m$, which proves Equation~\eqref{eq:ate-variance}.  Unbiased sample
variances on the two independent evaluation blocks therefore give the displayed
fixed-selected-coefficient variance estimator in Section~\ref{sec:ate}. Coverage
of its Wald interval additionally uses the stated evaluation-sample Lindeberg and
moment assumptions; variance unbiasedness alone is not a central limit theorem.

\subsection{Proof of Corollary~\ref{prop:ate-variance-quadratic}}

Expand the first variance in Equation~\eqref{eq:ate-variance} using
$Z_\lambda=Z_0+\lambda\Delta$. The interaction coefficient vanishes because
\begin{align}
\operatorname{Cov}_R(\Delta,\widehat{g})
&=
\mathbb E_R\left[
\{\widehat{g}-\mathbb E_R(\widehat{g})\}\mathbb E_R(\Delta\mid X)
\right]
 =0,
\notag\\
D
&=\operatorname{Cov}_R(Z_0,\widehat{g})
=\operatorname{Cov}_R\{\mathbb E_R(Z_0\mid X),\widehat{g}\}
=\operatorname{Cov}_R\{\tau(X),\widehat{g}(X)\}.
\label{eq:ate-population-separation}
\end{align}
Collecting the coefficient-dependent terms and multiplying by $n$ gives
$2C\lambda-2D\omega+A\lambda^2+B\omega^2$, which proves
Equation~\eqref{eq:ate-quadratic}. Finally, for every $(u,v)$,
\begin{align*}
\begin{pmatrix}u&v\end{pmatrix}
\mathbf H_{\mathrm{ATE}}
\begin{pmatrix}u\\v\end{pmatrix}
&=Au^2+Bv^2\geq0,
\end{align*}
so $\mathbf H_{\mathrm{ATE}}$ is positive semidefinite.

\subsection{Proof of Proposition~\ref{prop:ate-argmin-equivalence}}

Equation~\eqref{eq:ate-quadratic} is the positive affine transformation
\begin{align*}
\operatorname{Var}\{\widehat\theta_{r_0}(\lambda,\omega)\}
&=\frac1n\operatorname{Var}_R(Z_0)
+\frac1n\mathcal J_{\mathrm{ATE}}(\lambda,\omega),
\end{align*}
where $n>0$. Adding a constant and multiplying by a positive scalar preserve
the complete argmin set over every feasible set.

\subsection{Proof of Theorem~\ref{thm:joint-oracle}}

When $A,B>0$, the two first-order conditions are
\begin{align*}
2C+2A\lambda&=0,
&
-2D+2B\omega&=0.
\end{align*}
The separated quadratic is strictly convex, so their unique solution is
$\lambda^\star=-C/A$ and $\omega^\star=D/B$.
For every $\rho\geq0$, the ridge objective has gradient
\begin{align*}
\begin{pmatrix}
2C+2(A+\rho)\lambda\\
-2D+2(B+\rho)\omega
\end{pmatrix}.
\end{align*}
Since $A,B>0$, both $A+\rho$ and $B+\rho$ are positive for every
$\rho\geq0$. Setting the gradient to zero therefore proves
Equation~\eqref{eq:ridge-oracle} and uniqueness. At $\rho=0$, the penalized
objective reduces to $\mathcal J_{\mathrm{ATE}}$, whose minimizers are also
the minimizers of the exact population variance by
Proposition~\ref{prop:ate-argmin-equivalence}.

\subsection{Compact constraints}

Continuity of $\mathcal J_{\mathrm{ATE}}$ guarantees a minimizer on every nonempty compact
rectangle. If the rectangle contains the origin, its minimum variance cannot
exceed the RCT-only variance.

\subsection{Proof of Corollary~\ref{cor:joint-oracle-variance}}

For $A,B>0$, complete the two scalar squares:
\begin{align*}
\mathcal J_{\mathrm{ATE}}(\lambda,\omega)
&=
A\left(\lambda+\frac CA\right)^2
+B\left(\omega-\frac DB\right)^2
-\frac{C^2}{A}-\frac{D^2}{B}.
\end{align*}
The squared terms vanish at the oracle coefficients in
Equation~\eqref{eq:axis-oracle}. Substitution into
Equation~\eqref{eq:ate-quadratic} proves
Equation~\eqref{eq:ate-separable-minimum}.

\subsection{Prediction-powered CATE losses and finite-sample theory}


\paragraph{Proof of Proposition~\ref{prop:cate-target}.}
Fix $\zeta,\lambda,\omega$ and the fitted objects before using the tuning samples, and let $m$ be $m_\lambda$ or any fixed function. By Equation~\eqref{eq:pseudo-valid}, $\mathbb E_R\{Z_\lambda-\tau(X)\mid X\}=0$, so the cross term in $\mathbb E_R[\{Z_\lambda-\tau(X)\}+\{\tau(X)-\zeta(X)\}]^2$ vanishes and $\mathcal L_{\mathrm{DR}}(\zeta;\lambda)=Q_{\mathrm{DR}}(\lambda)+\mathcal R(\zeta)$. For the R loss, write $Y-m(X)-\{A-e(X)\}\zeta(X)=U+\{A-e(X)\}\{\tau(X)-\zeta(X)\}$ with $U\triangleq Y-m(X)-\{A-e(X)\}\tau(X)$. Conditional randomization gives $\mathbb E_R[\{A-e(X)\}\{Y-m(X)\}\mid X]=e(X)\{1-e(X)\}\tau(X)$ and $\mathbb E_R[\{A-e(X)\}^2\mid X]=e(X)\{1-e(X)\}$, hence $\mathbb E_R[U\{A-e(X)\}\mid X]=0$. Dividing the expanded square by $e(X)\{1-e(X)\}$ and taking $\mathbb E_R(\cdot\mid X)$ therefore leaves $\mathcal L_R(\zeta;m)=Q_R(m)+\mathcal R(\zeta)$. Finally, exact transport gives $\mathbb E_O\{r_0(X)G_\zeta(X)\}=\mathbb E_R\{G_\zeta(X)\}$, and $\mathbb E_R\{\chi(A,X)\mid X\}=1$ gives $\mathbb E_R\{\chi(A,X)G_\zeta(X)\}=\mathbb E_R\{G_\zeta(X)\}$, so both OBS corrections have mean zero for every $\omega$. This proves Equation~\eqref{eq:cate-target}.


\subsection{Proofs for coefficient selection}

For Proposition~\ref{prop:fusion-convexity}, expand the squares in Equations~\eqref{eq:dr-fusion-loss} and~\eqref{eq:rf-fusion-loss}. In the DRF loss, $\{Z_\lambda-\zeta(X)\}^2$ contributes $\zeta(X)^2$, and $G_\zeta(X)=\{\widehat g(X)-\zeta(X)\}^2$ contributes $\zeta(X)^2$ to each term of the OBS bracket, so the second-order part is $(1-\omega)\mathbb P_R^{\mathrm{tune}}\{\zeta(X)^2\}+\omega\mathbb P_O^{\mathrm{tune}}\{r(X)\zeta(X)^2\}$. In the RF loss, $[Y-m_\lambda(X)-\{A-e(X)\}\zeta(X)]^2/[e(X)\{1-e(X)\}]$ contributes $\chi(A,X)\zeta(X)^2$, which gives the RF part of Equation~\eqref{eq:fusion-curvature}. Since $\chi\geq0$, $r\geq0$, and $\omega\in[0,1]$, both parts are nonnegative, and a quadratic functional with a nonnegative second-order part is convex on any convex set. Taking expectations over the tuning samples with the fitted objects fixed replaces $\mathbb P_R^{\mathrm{tune}}$ and $\mathbb P_O^{\mathrm{tune}}$ by $\mathbb E_R$ and $\mathbb E_O$, which proves the claim for the expected losses.

For Proposition~\ref{prop:cate-noise-floor-lambda}, pseudo-outcome validity
gives $\mathbb E_R(\Delta\mid X)=0$.  Therefore
\begin{align*}
Q_{\mathrm{DR}}(\lambda)
&=Q_{\mathrm{DR}}(0)
+2\lambda\mathbb E_R[(Z_0-\tau)\Delta]
+\lambda^2\mathbb E_R(\Delta^2),\\
\mathbb E_R[(Z_0-\tau)\Delta]
&=\mathbb E_R(Z_0\Delta).
\end{align*}
Completing the square gives Equation~\eqref{eq:cate-dr-noise-lambda} when
$\mathbb E_R(\Delta^2)>0$.  If $\mathbb E_R(\Delta^2)=0$, then
$\Delta=0$ almost surely and the objective is flat.

For the R loss, write
$U_R\triangleq Y-m_R-(A-e)\tau$.  Since
$m_\lambda=m_R+\lambda d_m$,
\begin{align*}
Q_R(\lambda)
&=Q_R(0)
-2\lambda\mathbb E_R(U_Rd_m/\{e(1-e)\})
+\lambda^2\mathbb E_R(d_m^2/\{e(1-e)\}),\\
\mathbb E_R(U_Rd_m/\{e(1-e)\})
&=\mathbb E_R[\{Y-m_R\}d_m/\{e(1-e)\}].
\end{align*}
The second identity follows from
$\mathbb E_R(A-e\mid X)=0$.  Completing the square proves
Equation~\eqref{eq:cate-r-noise-lambda}.  If
$\mathbb E_R(d_m^2/\{e(1-e)\})=0$, strict overlap implies $d_m=0$ almost surely
and the objective is flat.  A convex
quadratic minimized over $[0,1]$ is minimized by projecting its
unrestricted minimizer onto that interval.


For Theorem~\ref{thm:cate-finite-validation}, condition on the fitted objects, which are independent of the evaluation samples by Assumption~\ref{ass:honest-separation}. Item 1: by Equation~\eqref{eq:pseudo-valid}, $\mathbb E_R\{Z_0-\widehat\zeta_j(X;\lambda,\omega)\}^2=Q_{\mathrm{DR}}(0)+\mathcal R\{\widehat\zeta_j(\cdot;\lambda,\omega)\}$ as in the proof of Proposition~\ref{prop:cate-target}, and exact transport gives $\mathbb E_O(r_0G_{j,\lambda,\omega})=\mathbb E_R(G_{j,\lambda,\omega})$, so the correction has mean $\omega\mathbb E_O(rG_{j,\lambda,\omega})-\omega\mathbb E_R(G_{j,\lambda,\omega})=\omega\mathbb E_O[(r-r_0)G_{j,\lambda,\omega}]$. Item 2: each RCT summand $\{Z_0-\widehat\zeta_j\}^2-\omega G_{j,\lambda,\omega}$ lies in $[-4B^2,4B^2]$ and each OBS summand $\omega rG_{j,\lambda,\omega}$ in $[0,4\bar rB^2]$, so Hoeffding's inequality bounds the deviation of the RCT average by $8B^2\sqrt{\log(4M/\alpha)/(2n)}$ and that of the OBS average by $4\bar rB^2\sqrt{\log(4M/\alpha)/(2N)}$, each with probability at least $1-\alpha/(2M)$. A union bound over the $M$ grid pairs and the two averages gives both bounds for every pair with probability at least $1-\alpha$, and the two deviations sum to $\varepsilon_{\mathrm{eval}}(\alpha)$. Item 3: write $Q$ for $Q_{\mathrm{DR}}(0)$, $b(\lambda,\omega)$ for the transport error in item 1, $\widehat s$ for the selected pair, and $s^\star$ for a grid pair minimizing the CATE risk. On the event of item 2, $\mathcal R(\widehat s)=\mathbb E\{\widehat C_j(\widehat s)\}-Q-b(\widehat s)\leq\widehat C_j(\widehat s)-Q-b(\widehat s)+\varepsilon_{\mathrm{eval}}(\alpha)\leq\widehat C_j(s^\star)-Q-b(\widehat s)+\varepsilon_{\mathrm{eval}}(\alpha)\leq\mathcal R(s^\star)+b(s^\star)-b(\widehat s)+2\varepsilon_{\mathrm{eval}}(\alpha)$, where the middle inequality holds because $\widehat s$ minimizes $\widehat C_j$ over the grid. Bounding $|b(s^\star)-b(\widehat s)|$ by twice the maximum of $|b|$ over the grid proves Equation~\eqref{eq:cate-selector-inequality}.


\subsection{Proofs for the orthogonality audit}


Condition throughout on the nuisance samples, so that $\widehat\mu_R$, $\widehat\mu_O$, $\widehat g$, $r$, $\lambda$, and $\omega$ are fixed before the tuning samples are used, and assume enough integrability to interchange differentiation and expectation.

\paragraph{Learning equations.}
Differentiating the expected losses of Definition~\ref{def:fusion-losses} in the direction $v$ and multiplying by $-\tfrac12$ gives Equations~\eqref{eq:drf-learning-equation} and~\eqref{eq:rf-learning-equation}. For $r=r_0$, exact transport gives $\mathbb E_O(r_0f)=\mathbb E_R(f)$ for each integrable $f(X)$, and Assumption~\ref{ass:trial} gives $\mathbb E_R\{\chi(A,X)\mid X\}=1$, so the two terms of each OBS bracket cancel, which gives the reduced forms displayed after Definition~\ref{def:learning-equations}.

\paragraph{Proof of Proposition~\ref{prop:cate-orthogonality-audit}.}
For a pair $h=(h_0,h_1)$ write $c(h)\triangleq\{A-e(X)\}\{h_1(X)/e(X)+h_0(X)/[1-e(X)]\}$. Differentiating $\varphi(V;q)$ in Definition~\ref{def:rct-aipw-score} along $q+\varepsilon h$ gives $-c(h)$, and $\mathbb E_R\{c(h)\mid X\}=0$ by conditional randomization. Along $\widehat\mu_R+\varepsilon h$, the fused pair $\widehat\mu_\lambda$ moves by $(1-\lambda)h$ and $m_\lambda$ moves by $(1-\lambda)\{e(X)h_1(X)+[1-e(X)]h_0(X)\}$, while $\widehat g$ does not move. Hence
\begin{align*}
D_{\widehat\mu_R}\Psi_{\mathrm{DRF}}[h,v]&=-(1-\lambda)\mathbb E_R\{v(X)c(h)\}=0,
\\
D_{\widehat\mu_R}\Psi_{\mathrm{RF}}[h,v]&=-(1-\lambda)\mathbb E_R\left[v(X)\frac{A-e(X)}{e(X)\{1-e(X)\}}\{e(X)h_1(X)+[1-e(X)]h_0(X)\}\right]=0,
\end{align*}
because $v$ and $h$ are functions of $X$ and $\mathbb E_R\{A-e(X)\mid X\}=0$. This proves item 1. Along $\widehat\mu_O+\varepsilon h$, the same two contributions appear with $\lambda$ in place of $1-\lambda$ and vanish for the same reason, and in addition $\widehat g$ moves by $u=h_1-h_0$. Only the OBS bracket depends on $\widehat g$, and its derivative is $\omega[\mathbb E_O\{r(X)v(X)u(X)\}-\mathbb E_R\{v(X)u(X)\}]$ for DRF and $\omega[\mathbb E_O\{r(X)v(X)u(X)\}-\mathbb E_R\{\chi(A,X)v(X)u(X)\}]$ for RF. Exact transport and $\mathbb E_R\{\chi(A,X)\mid X\}=1$ turn both into $\omega\mathbb E_O[\{r(X)-r_0(X)\}u(X)v(X)]$, which proves item 2. Along $r+\varepsilon h_r$, only the positive OBS term moves, so $D_r\Psi_j[h_r,v]=\omega\mathbb E_O[h_r(X)v(X)\{\widehat g(X)-\zeta(X)\}]$. For the last claim of item 3, take $f\triangleq v(\widehat g-\zeta)\in L_2(P_O^X)$. Its inner product with every mean-zero element of $L_2(P_O^X)$ is zero if and only if $f$ lies in the orthogonal complement of the mean-zero functions, which consists of the almost-sure constants; for example, the centered direction $h_r=f-\mathbb E_O(f)$ has pairing $\mathbb E_O(h_rf)=\operatorname{Var}_O(f)$. Multiplication by nonzero $\omega$ does not change the zero set.


\subsection{Proofs for the linear sieve}
\label{app:sieve-proofs}

For Proposition~\ref{prop:sieve-closed-form}, item 1: for DRF the two displays coincide. For RF, $\chi(A,X)\{\widetilde Z_{\mathrm{RF},\lambda}-\zeta(X)\}^2=\{A-e(X)\}^2\left[\{Y-m_\lambda(X)\}/\{A-e(X)\}-\zeta(X)\right]^2/[e(X)\{1-e(X)\}]=[Y-m_\lambda(X)-\{A-e(X)\}\zeta(X)]^2/[e(X)\{1-e(X)\}]$, which is the first term of Equation~\eqref{eq:rf-fusion-loss}, and the OBS brackets agree by the definition of $w_j$. Item 3: substitute $\widetilde Z_{j,\lambda}=\widetilde Z_{j,0}+\lambda\widetilde\Delta_j$ into the right-hand side of Equation~\eqref{eq:sieve-normal-equation} and multiply by the inverse.

For Theorem~\ref{thm:sieve-risk}, item 1: $\mathbb E_R\{b(\tau-\zeta_p)\}=\mathbb E_R(b\tau)-\Gamma_R\beta_p=0$, so expanding $\mathcal R(b^\top\beta)=\mathbb E_R[b^\top(\beta-\beta_p)+\{\zeta_p(X)-\tau(X)\}]^2$ leaves no cross term. By item 1, $\mathbb E\{\mathcal R(\widehat\zeta_j)\}-a_p^2$ is the expected squared $\Gamma_R$-norm of this sum. The leading term has mean zero by item 2, so its expected squared $\Gamma_R$-norm equals $\operatorname{tr}[\Gamma_R^{-1}\operatorname{Var}\{\mathbb P_R^{\mathrm{tune}}\psi_{R,j}+\mathbb P_O^{\mathrm{tune}}\psi_O\}]=\mathcal J_j(\lambda,\omega)$ by Equation~\eqref{eq:sieve-score-variance}; the cross term is $o(1/n_R^{\mathrm{tune}}+1/n_O^{\mathrm{tune}})$ by the Cauchy--Schwarz inequality and the remainder condition, and the squared remainder is of the same smaller order.

For Corollary~\ref{cor:sieve-oracle}, $\psi_{R,j}(\lambda,\omega)=w_jb(\widetilde Z_{j,0}-\zeta_p)+\lambda\,w_jb\widetilde\Delta_j-\omega\,w_jb(\widehat g-\zeta_p)$ and $\psi_O(\omega)=\omega r_0b(\widehat g-\zeta_p)$. For DRF, $w_j=1$ and $\mathbb E_R(\Delta\mid X)=0$, so it vanishes; for RF, $\mathbb E_R(\chi^2\widetilde\Delta_{\mathrm{RF}}\mid X)=-\{m_O(X)-m_R(X)\}\{1-2e(X)\}/[e(X)\{1-e(X)\}]$, which vanishes when $e\equiv1/2$. For DRF the minimizer, its projection, and the improvement follow by completing the two squares, as in the proof of Theorem~\ref{thm:joint-oracle}.

\subsection{A two-stratum oracle calculation}

Let $P_R(X=0)=P_R(X=1)=1/2$, $\tau(0)=0$, $\tau(1)=2$, and $\widehat{g}=\tau$. With $n=100$, $N=900$, and a common covariate marginal,
\begin{align*}
\mathbb E_R\{\tau(X)\}&=1,
&
\operatorname{Var}_R\{\tau(X)\}&=1,
&
\omega_{\mathrm{ATE}}^\star&=\frac{1}{1+100/900}=0.9.
\end{align*}
The baseline between-$X$ variance is $0.01$, whereas the oracle component is
\begin{align*}
\frac{\operatorname{Var}_R\{0.1\tau(X)\}}{100}
+\frac{0.9^2\operatorname{Var}_R\{\tau(X)\}}{900}
&=0.0001+0.0009=0.001.
\end{align*}
The reduction $0.009$ concerns only between-$X$ variation. The first RCT
variance term in Equation~\eqref{eq:ate-variance} also contains conditional
pseudo-outcome noise.

\subsection{Honest variance estimation and evaluation-sample inference}
\label{app:ate-evaluation-regularity}

For Proposition~\ref{prop:ate-honest-variance-active}, fix the coefficients and
scores before using the evaluation observations. Ordinary unbiased sample-variance identities
give
\begin{align*}
\mathbb E\{\widehat{\operatorname{Var}}_{R,\mathrm{eval}}(\psi_R)\}
&=\operatorname{Var}_R(\psi_R),
&
\mathbb E\{\widehat{\operatorname{Var}}_{O,\mathrm{eval}}(\psi_O)\}
&=\operatorname{Var}_O(\psi_O).
\end{align*}
Dividing by $n$ and $N$, then adding, proves unbiasedness of
$\widehat V$ by independence of $D_R^{\mathrm{eval}}$ and
$D_O^{\mathrm{eval}}$.

For Theorem~\ref{thm:ate-honest-clt-active}, impose the following evaluation
regularity conditions after fixing the learned objects:
\begin{enumerate}
\item[(R1)] the RCT and OBS evaluation observations are conditionally
independent and retain their respective product laws;
\item[(R2)] $n,N\to\infty$, $n/N\to\kappa\in[0,\infty)$, and for some
$\delta>0$ the centered RCT and OBS evaluation-score arrays have uniformly
bounded conditional $(2+\delta)$ moments;
\item[(R3)]
$n\operatorname{Var}\{\widehat\theta_r(\widehat\lambda,
\widehat\omega)\}\to\sigma^2\in(0,\infty)$ in probability, and the combined
sample-variance estimator is ratio-consistent for this variance.
\end{enumerate}
Center the two evaluation-score arrays at their population means. Their
weighted sum is $\widehat\theta_r-\theta$. Condition (R2) supplies the
two-sample triangular-array central limit theorem: the RCT Lyapunov remainder
is of order $n^{-\delta/2}$, and the OBS remainder also vanishes when
$n/N\to\kappa<\infty$. Condition (R3) and Slutsky's theorem yield the
studentized limit and asymptotic coverage of the normal-approximation interval.

\subsection{Proof of Theorem~\ref{thm:ate-three-way-comparison}}
\label{app:ate-three-way-comparison}

Under either exact $r=r_0$ transport or its common-marginal special case,
Equation~\eqref{eq:pseudo-valid} gives
$\mathbb E_R(\Delta\mid X)=0$. Hence
$\operatorname{Cov}_R(\Delta,\widehat{g})=0$, as shown in
Equation~\eqref{eq:ate-population-separation}. Multiplying
Equation~\eqref{eq:ate-quadratic} by $n$ therefore gives
\begin{align*}
\widetilde{\mathcal V}(\lambda,\omega)
&=
\sigma_0^2+2C\lambda-2D\omega+A\lambda^2+B\omega^2.
\end{align*}
At $(\lambda,\omega)=(0,0)$, this equals $\sigma_0^2$. Holding
$\lambda=0$ and completing the square in $\omega$ gives the unique
PPI\textsuperscript{++}-only oracle $D/B$ and variance $\sigma_0^2-D^2/B$. Completing the
two separate squares gives the joint oracle $(-C/A,D/B)$ and variance
$\sigma_0^2-C^2/A-D^2/B$. The two variance gaps are $D^2/B$ and
$C^2/A$. Because $A,B>0$, they are nonnegative and vanish exactly when
$D=0$ and $C=0$, respectively.

\subsection{Density-ratio characterizations and estimated-ratio inheritance}
\label{app:estimated-ratio-inheritance}

\paragraph{Proof of Proposition~\ref{prop:ratio-characterizations}.}
The defining property of the Radon--Nikodym derivative gives
\begin{align*}
\mathbb E_O\{r_0(X)h(X)\}
&=
\mathbb E_R\{h(X)\}
\end{align*}
for every integrable $h$. If a nonnegative normalized function
$\widetilde r$ gives
the same equality for every bounded measurable $h$, then
$\int_A \widetilde r\,dP_O=P_R(A)=\int_A r_0\,dP_O$ for every measurable
set $A$. Uniqueness of the Radon--Nikodym derivative gives
$\widetilde r=r_0$
$P_O^X$-almost surely.

\paragraph{Proof of Proposition~\ref{prop:ratio-riesz-characterization}.}
If $r_0\in L_2(P_O^X)$, then for every $a\in L_2(P_O^X)$,
\begin{align*}
\mathcal R(a)
&=
\mathbb E_O\{a(X)^2\}-2\mathbb E_R\{a(X)\}
\\
&=
\mathbb E_O\{(a(X)-r_0(X))^2\}
-\mathbb E_O\{r_0(X)^2\}.
\end{align*}
The squared term is uniquely minimized at $a=r_0$ in
$L_2(P_O^X)$. Restricting the balance equations to the coordinates of
$\phi$ tests equality only against functions in their linear span, which
proves the finite-feature qualification.

\paragraph{Proof of Lemma~\ref{lem:linear-riesz-ratio}.}
For $a_\gamma=\phi^\top\gamma$, the gradient of the objective in
Equation~\eqref{eq:ratio-riesz-empirical} is twice
\begin{align*}
\left[
\mathbb P_O^{\mathrm{nuis}}\{\phi(X)\phi(X)^\top\}+\rho I_p
\right]\gamma
-\mathbb P_R^{\mathrm{nuis}}\{\phi(X)\}.
\end{align*}
The stated invertibility condition makes its unique zero
Equation~\eqref{eq:ratio-riesz-linear}.

\paragraph{Classifier odds.}
Let $p_R$ and $p_O$ be densities with respect to a common dominating
measure. In the pooled nuisance population,
\begin{align*}
s_0(x)
&=
\frac{\pi p_R(x)}{\pi p_R(x)+(1-\pi)p_O(x)}.
\end{align*}
Solving for $p_R(x)/p_O(x)$ gives
\begin{align*}
r_0(x)
&=
\frac{1-\pi}{\pi}\frac{s_0(x)}{1-s_0(x)},
\end{align*}
which proves Equation~\eqref{eq:ratio-classifier-bayes} in
Definition~\ref{def:classifier-density-ratio}. The same algebra with a fitted
domain probability and the effective training prior yields
$\widehat r_{\mathrm{CLS}}$.

\paragraph{Proof of Proposition~\ref{prop:automatic-ratio-balance}.}
The log of a positive weighted exponential sum is convex in $\xi$. At an
interior minimizer, differentiating the objective in
Definition~\ref{def:classifier-calibrated-balance} gives
\begin{align*}
0
&=
\frac{
\mathbb P_O^{\mathrm{nuis}}
\{\widehat r_{\mathrm{base}}(X)e^{\widehat\xi^\top\phi(X)}\phi(X)\}
}{
\mathbb P_O^{\mathrm{nuis}}
\{\widehat r_{\mathrm{base}}(X)e^{\widehat\xi^\top\phi(X)}\}
}
-\mathbb P_R^{\mathrm{nuis}}\{\phi(X)\}
\\
&=
\mathbb P_O^{\mathrm{nuis}}
\{\widehat r_{\mathrm{BAL}}(X)\phi(X)\}
-\mathbb P_R^{\mathrm{nuis}}\{\phi(X)\}.
\end{align*}
The denominator normalizes the OBS weights to one, and positivity follows from
the positive base ratio and exponential tilt. A finite interior optimizer can
exist only when the target feature mean lies in the relative interior of the
convex hull of the weighted OBS feature vectors. Boundary or infeasible cases require a separate
approximate-balance rule and are not covered by the exact first-order
condition.


\paragraph{Proof of Proposition~\ref{prop:estimated-ratio-effect}.}
Item 1. On the tuning OBS sample, $\widehat r\widehat g=r_0\widehat g+(\widehat r-r_0)\widehat g$, and the sample-variance identity $\widehat{\operatorname{Var}}(U+V)=\widehat{\operatorname{Var}}(U)+2\widehat{\operatorname{Cov}}(U,V)+\widehat{\operatorname{Var}}(V)$ with $U=r_0\widehat g$ and $V=(\widehat r-r_0)\widehat g$ proves Equation~\eqref{eq:ratio-B-error}. The moments $\widehat A$, $\widehat C$, and $\widehat D$ in Equation~\eqref{eq:ate-empirical-moments} do not involve the ratio, so the ratio enters Equation~\eqref{eq:ate-separated-plugin} only through $\widehat B$.

Item 2. At the same selected coefficients, the two estimators in Equation~\eqref{eq:ate-estimator} share the term $\mathbb P_{R,n}Z_{\widehat\lambda}-\widehat\omega\mathbb P_{R,n}\widehat g$ and differ only in $\widehat\omega\mathbb P_{O,N}(r\widehat g)$, whose difference across $r=\widehat r$ and $r=r_0$ is $\widehat\omega\mathbb P_{O,N}\{(\widehat r-r_0)\widehat g\}$.

Item 3. Condition on all fitted and selected objects. With the evaluation scores of Algorithm~\ref{alg:honest-ate}, $\widehat\theta_{\widehat r}=\mathbb P_{R,n}\psi_R+\mathbb P_{O,N}\psi_O$. The exact drift identity in Equation~\eqref{eq:ate-ratio-drift} gives the conditional mean $\theta+\widehat\omega\,\mathbb E_O\{(\widehat r-r_0)\widehat g\}$, and the transport identity in Equation~\eqref{eq:ratio-transport-characterization} rewrites $\mathbb E_O(r_0\widehat g)$ as $\mathbb E_R(\widehat g)$. Independence of the two evaluation samples gives $\operatorname{Var}(\widehat\theta_{\widehat r})=\operatorname{Var}_R(\psi_R)/n+\operatorname{Var}_O(\psi_O)/N$, and each sample variance in $\widehat V$ is conditionally unbiased for the corresponding variance, which proves Equation~\eqref{eq:estimated-ratio-conditional-identities}.

\paragraph{Proof of Theorem~\ref{thm:estimated-ratio-feasible-clt}.}
Write $\beta\triangleq\widehat\omega\{\mathbb E_O(\widehat r\widehat g)-\mathbb E_R(\widehat g)\}$ for the drift in item 3 of Proposition~\ref{prop:estimated-ratio-effect}. Continue conditioning on the fitted and selected objects. Center the two
evaluation scores at their respective conditional means. The resulting two
independent triangular arrays sum to
$\widehat\theta_{\widehat r}-(\theta+\beta)$ and have total variance
$V_n$. On the event
$nV_n\geq\sigma^2/2$, whose probability tends to one, the uniform
conditional $(2+\delta)$-moment bound and bounded $n/N$ make the conditional
Lyapunov ratio bounded by an $O_p(1)$ factor times
\begin{align*}
n^{-\delta/2}+\frac{n^{1+\delta/2}}{N^{1+\delta}}
&=o_p(1).
\end{align*}
Thus its conditional law given the fitted and selected objects converges in
probability to $N(0,1)$ after division by $\sqrt{V_n}$. This also
implies the unconditional convergence.

It remains to establish rather than assume variance consistency. Let
$\eta\triangleq\min(\delta/2,1)$. For a centered score $\xi$, the conditional
$(2+\delta)$-moment bound gives
$\mathbb E(|\xi^2-\mathbb E\xi^2|^{1+\eta}\mid\text{fitted objects})=O_p(1)$.
The von Bahr--Esseen inequality therefore bounds the conditional
$(1+\eta)$-moment of the centered empirical second moment by
$O_p(m^{-\eta})$ for a sample of size $m$. Moreover, the squared empirical
mean is $o_p(1)$ because its conditional expectation is $O_p(m^{-1})$.
Consequently, for $S\in\{R,O\}$,
\begin{align*}
\widehat{\operatorname{Var}}_{S,\mathrm{eval}}(\psi_S)
-\operatorname{Var}_S(\psi_S)
&=o_p(1).
\end{align*}
Consequently, because $n/N$ is bounded,
\begin{align*}
n\{\widehat V-V_n\}=o_p(1)+\frac nN o_p(1)=o_p(1).
\end{align*}
Together with $nV_n\to_p\sigma^2>0$, this proves
$\widehat V/V_n\to_p1$. Finally,
\begin{align*}
\frac{\widehat\theta_{\widehat r}-\theta}{\sqrt{\widehat V}}=\frac{\widehat\theta_{\widehat r}-(\theta+\beta)}{\sqrt{V_n}}\sqrt{\frac{V_n}{\widehat V}}+\frac{\beta}{\sqrt{V_n}}\sqrt{\frac{V_n}{\widehat V}}.
\end{align*}
The bias condition and Slutsky's theorem prove the stated limit and coverage.


\paragraph{Proof of Corollary~\ref{cor:estimated-ratio-inheritance}.}
Fix $k\in\{\mathrm{RCT},\mathrm{PPI}^{++},\mathrm{joint}\}$ and write $(\widehat\lambda,\widehat\omega)\triangleq(\widehat\lambda_k,\widehat\omega_k)$ and $(\lambda^\star,\omega^\star)\triangleq(\lambda_k^\star,\omega_k^\star)$. Decompose
\begin{align*}
\widehat\theta_{\widehat r,k}-\widehat\theta_{r_0}(\lambda^\star,\omega^\star)
&=\left\{\widehat\theta_{\widehat r}(\widehat\lambda,\widehat\omega)-\widehat\theta_{r_0}(\widehat\lambda,\widehat\omega)\right\}
+\left\{\widehat\theta_{r_0}(\widehat\lambda,\widehat\omega)-\widehat\theta_{r_0}(\lambda^\star,\omega^\star)\right\}.
\end{align*}
By item 2 of Proposition~\ref{prop:estimated-ratio-effect}, the first difference equals $\widehat\omega\mathbb P_{O,N}\{(\widehat r-r_0)\widehat g\}$. Its conditional mean is the drift of item 3, which the bias condition in Theorem~\ref{thm:estimated-ratio-feasible-clt} makes $o_p(n^{-1/2})$ because $nV_n=O_p(1)$. Its conditional variance is $\widehat\omega^2\operatorname{Var}_O\{(\widehat r-r_0)\widehat g\}/N$, so Equation~\eqref{eq:estimated-ratio-fluctuation-rate} and conditional Chebyshev make its centered part $o_p(n^{-1/2})$. For the second difference,
\begin{align*}
\widehat\theta_{r_0}(\widehat\lambda,\widehat\omega)-\widehat\theta_{r_0}(\lambda^\star,\omega^\star)
&=(\widehat\lambda-\lambda^\star)\mathbb P_{R,n}\Delta
+(\widehat\omega-\omega^\star)\left\{\mathbb P_{O,N}(r_0\widehat g)-\mathbb P_{R,n}\widehat g\right\}.
\end{align*}
RCT validity gives $P_R\Delta=0$, and exact transport gives $P_O(r_0\widehat g)=P_R(\widehat g)$. Together with the second-moment bounds in Equation~\eqref{eq:estimated-ratio-stability} and bounded $n/N$, these identities make both empirical factors $O_p(n^{-1/2})$, and coefficient consistency makes the second difference $o_p(n^{-1/2})$. Combining the two differences proves Equation~\eqref{eq:estimated-ratio-first-order-equivalence}. Convergence of the exact-ratio second moments then transfers the ordering in Theorem~\ref{thm:ate-three-way-comparison}, which is Equation~\eqref{eq:estimated-ratio-inherited-ordering}.

\subsection{Counterexamples to adaptive reuse}

The fixed-coefficient theorem cannot be extended by selecting a coefficient on
the same evaluation outcomes without a separate analysis.

\paragraph{Reuse through $\lambda$.}
Let $X$ be degenerate, $\theta=0$, $Z_0=0$, and let the $\Delta_i$ be i.i.d. Rademacher variables. Every fixed $\lambda$ satisfies $\mathbb E\{\lambda\mathbb P_{R,n}\Delta\}=0$. If the same outcomes select $\widehat\lambda=\operatorname{sign}(\mathbb P_{R,n}\Delta)$, then
\begin{align*}
\mathbb P_{R,n}(Z_0+\widehat\lambda\Delta)
&=|\mathbb P_{R,n}\Delta|,
\end{align*}
whose expectation is strictly positive.

\paragraph{Reuse through $\omega$.}
Under the common covariate marginal, let $\widehat{g}(X)$ be a nondegenerate mean-zero Rademacher variable and let the RCT-valid score be identically zero. Every fixed $\omega$ centers $\omega\{\mathbb P_{O,N}\widehat{g}-\mathbb P_{R,n}\widehat{g}\}$. Choosing $\widehat\omega$ as the sign of that same bracket produces its absolute value, which again has strictly positive expectation. The bias in both examples comes solely from adaptive reuse.
