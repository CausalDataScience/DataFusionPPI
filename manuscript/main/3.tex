\section{Prediction-Powered Fusion for the ATE}
\label{sec:ate}

Throughout this section, Assumptions~\ref{ass:trial} and
\ref{ass:honest-separation} hold, the evaluation sample sizes satisfy $n,N>0$,
and either Assumption~\ref{ass:common-marginal} or
Assumption~\ref{ass:exact-transport} holds. Thus, the exact density ratio $r_0$
in Definition~\ref{def:true-density-ratio} exists and equals one under the
common-covariate regime. A feasible estimator may use a fixed candidate weight
$r$; results that require the exact weight state $r=r_0$ explicitly.

\subsection{Estimator, exact variance, and population oracles}

\begin{definition}[ATE fusion estimator and its components]
	Using the fitted OBS outcome regression $\widehat\mu_O$ from
	Equation~\eqref{eq:obs-outcome-regression}, define the fitted observational
	treatment-effect prediction by
	\begin{align}
		\widehat{g}(x)
		&\triangleq \widehat\mu_{O,1}(x)-\widehat\mu_{O,0}(x).
		\label{eq:obs-effect-prediction}
	\end{align}
	Let $r:\mathcal X\to[0,\infty)$ be a fixed measurable candidate transport
	weight. For fixed $(\lambda,\omega)$, define the ATE fusion estimator by
	\begin{align}
		\widehat\theta_r(\lambda,\omega) &\triangleq
		\mathbb P_{R,n}Z_\lambda
		+
		\omega\left\{
		\mathbb P_{O,N}\{r(X)\widehat{g}(X)\}
		-
		\mathbb P_{R,n}\widehat{g}(X)
		\right\},
		\label{eq:ate-estimator}
	\end{align}
	where $Z_\lambda$ is the RCT-valid score in
	Equation~\eqref{eq:fusion-score}. The fitted outcome regressions
	$\widehat\mu_R$ and $\widehat\mu_O$ are defined in
	Equations~\eqref{eq:rct-outcome-regression} and
	\eqref{eq:obs-outcome-regression}, respectively. The coefficient $\lambda$
	blends these regressions inside $Z_\lambda$, whereas $\omega$ scales the
	transported control variate. Exact transport uses $r=r_0$.
\end{definition}
At the coefficient origin,
\begin{align*}
	\widehat\theta_r(0,0) &=\mathbb P_{R,n}Z_0 =\widehat\theta_{\mathrm{AIPW},R},
\end{align*}
where $\widehat\theta_{\mathrm{AIPW},R}$ is defined in Equation~\eqref{eq:rct-only-aipw}, so the family contains the RCT-only AIPW estimator.

The estimator depends on three learned components: the outcome-regression blend inside $Z_\lambda \triangleq \varphi(V;\widehat{\mu}_{\lambda})$, the OBS prediction $\widehat{g}$, and the density ratio $r$. The next result states exactly which components can change its population mean.

\begin{theorem}[Exact target, density-ratio drift, and variance]
\label{prop:ate-nuisance-sensitivity}
	\label{thm:ate-variance}
	\label{thm:ate-exact-r}
	For every fixed $(\lambda,\omega)$, arbitrary fixed outcome-regression fits
	and a fixed OBS prediction satisfy, whenever the displayed expectations are
	finite,
	\begin{align}
		\mathbb E\{\widehat\theta_{r_0}(\lambda,\omega)\} &=\theta.
		\label{eq:ate-exact-target}
	\end{align}
	For any fixed candidate weight $\widetilde r$, its exact mean drift is
	\begin{align}
		\mathbb E\{\widehat\theta_{\widetilde r}(\lambda,\omega)\}-\theta
		&=\omega\mathbb E_O\left[\{\widetilde r(X)-r_0(X)\}\widehat{g}(X)\right].
		\label{eq:ate-ratio-drift}
	\end{align}
	If $Z_\lambda-\omega\widehat{g}(X)$ and
	$r_0(X)\widehat{g}(X)$ are square-integrable, then
	\begin{align}
		\operatorname{Var}\{\widehat\theta_{r_0}(\lambda,\omega)\}
		&=
		\frac1n\operatorname{Var}_R\{Z_\lambda-\omega \widehat{g}(X)\}
		+
		\frac{\omega^2}{N}\operatorname{Var}_O\{r_0(X)\widehat{g}(X)\}.
		\label{eq:ate-variance}
	\end{align}
\end{theorem}
Equation~\eqref{eq:ate-exact-target} shows that, under exact transport
$r=r_0$, the expectation of $\widehat\theta_{r_0}(\lambda,\omega)$ is
unchanged by the fixed outcome regressions $\widehat\mu_R$ and
$\widehat\mu_O$ or by the fixed OBS prediction $\widehat{g}$. If a candidate
weight $\widetilde r$ differs from $r_0$, Equation~\eqref{eq:ate-ratio-drift}
gives the resulting mean error exactly.

We next minimize the exact variance in Equation~\eqref{eq:ate-variance} over the two fusion coefficients.

\begin{definition}[ATE variance coefficients]
\label{def:ate-variance-objects}
Recall from Equation~\eqref{eq:fusion-score} and
Section~\ref{sec:rct-obs-setting} that
$Z_0\triangleq\varphi(V;\widehat\mu_R)$ and $\Delta\triangleq Z_1-Z_0$;
$\widehat{g}$ is the OBS prediction in
Equation~\eqref{eq:obs-effect-prediction}. Define
	\begin{align}
		A&\triangleq\operatorname{Var}_R(\Delta),
		&
		B&\triangleq\operatorname{Var}_R(\widehat{g})+\frac nN\operatorname{Var}_O(r_0\widehat{g}),
		\notag\\
		C&\triangleq\operatorname{Cov}_R(Z_0,\Delta),
		&
		D&\triangleq\operatorname{Cov}_R(Z_0,\widehat{g}).
	\label{eq:ate-moments}
	\end{align}
	Here $A$ and $B$ are the quadratic costs of changing $\lambda$ and
	$\omega$, while $C$ and $D$ determine their linear slopes.
Also define
\begin{align*}
	\mathbf H_{\mathrm{ATE}}
	&\triangleq
	\begin{pmatrix}A&0\\0&B\end{pmatrix},
	&
	\boldsymbol\ell
	&\triangleq
	\begin{pmatrix}C\\-D\end{pmatrix},
	&
	\boldsymbol\beta
	&\triangleq
	\begin{pmatrix}\lambda\\\omega\end{pmatrix}.
\end{align*}
\end{definition}

The following corollary rewrites the variance in Equation~\eqref{eq:ate-variance} using the four coefficients $A$, $B$, $C$, and $D$.

\renewcommand*{\theHtheoremcorollary}{\theHtheorem.\arabic{theoremcorollary}}
\begin{theoremcorollary}[Separated ATE variance quadratic]
\label{prop:ate-variance-quadratic}
If the displayed scores have finite second moments, then
\begin{align}
	\operatorname{Var}\{\widehat\theta_{r_0}(\lambda,\omega)\}
	&=
	\frac1n\left\{
	\operatorname{Var}_R(Z_0)+2C\lambda-2D\omega
	+A\lambda^2+B\omega^2
	\right\}.
	\label{eq:ate-quadratic}
\end{align}
\end{theoremcorollary}

Removing the constant term from Equation~\eqref{eq:ate-quadratic} gives the following equivalent optimization objective.

\begin{proposition}[Equivalent ATE coefficient objective]
\label{def:ate-coefficient-objective}
\label{prop:ate-argmin-equivalence}
	Define
	\begin{align*}
		\mathcal J_{\mathrm{ATE}}(\lambda,\omega) &\triangleq 2C\lambda-2D\omega+A\lambda^2+B\omega^2.
	\end{align*}
	For every nonempty compact $\Theta\subseteq\mathbb R^2$,
	\begin{align*}
		\operatorname*{arg\,min}_{(\lambda,\omega)\in\Theta} \operatorname{Var}\{\widehat\theta_{r_0}(\lambda,\omega)\} &=
		\operatorname*{arg\,min}_{(\lambda,\omega)\in\Theta}
		\mathcal J_{\mathrm{ATE}}(\lambda,\omega).
	\end{align*}
\end{proposition}

The next theorem gives the closed-form variance-minimizing coefficients and their ridge-regularized counterparts.

\begin{theorem}[Population and ridge oracles]
\label{def:ate-joint-ridge-oracles}
\label{thm:joint-oracle}
	Suppose the displayed scores have finite second moments and $A,B>0$. For
	every $\rho\geq0$, the unique ridge optimizer is
	\begin{align}
		\begin{pmatrix}
			\lambda_\rho^\star\\
			\omega_\rho^\star
		\end{pmatrix}
		&\triangleq
		\operatorname*{arg\,min}_{(\lambda,\omega)\in\mathbb R^2}
		\left\{\mathcal J_{\mathrm{ATE}}(\lambda,\omega)
		+\rho(\lambda^2+\omega^2)\right\}
		=
		\begin{pmatrix}
			-C/(A+\rho)\\
			D/(B+\rho)
		\end{pmatrix}.
		\label{eq:ridge-oracle}
	\end{align}
	At $\rho=0$, this is the unique minimizer of the exact population variance:
	\begin{align}
		\lambda^\star&\triangleq\lambda_0^\star=-\frac CA,
		&
		\omega^\star&\triangleq\omega_0^\star=\frac DB.
		\label{eq:axis-oracle}
	\end{align}
\end{theorem}

\begin{theoremcorollary}[Minimum population variance]
\label{cor:joint-oracle-variance}
\label{cor:ate-separable-oracle}
	Suppose the displayed scores have finite second moments and $A,B>0$. Define
\begin{align}\label{eq:ate-minimum-definition}
	\mathcal V^\star
	&\triangleq
	\min_{(\lambda,\omega)\in\mathbb R^2}
	\operatorname{Var}\{\widehat\theta_{r_0}(\lambda,\omega)\}.
\end{align}
Let $\mathcal V_0\triangleq\operatorname{Var}_R(Z_0)/n$. Then
\begin{align}\label{eq:ate-separable-minimum}
	\mathcal V^\star
	&=
	\mathcal V_0-
	\frac1n\left(\frac{C^2}{A}+\frac{D^2}{B}\right),
	&
	\mathcal V_0-\mathcal V^\star
	&=
	\frac1n\left(\frac{C^2}{A}+\frac{D^2}{B}\right)\geq0.
\end{align}
\end{theoremcorollary}

The next theorem separates the two sources of this oracle variance reduction.

\begin{theorem}[Exact-ratio oracle variance comparison]
\label{thm:ate-three-way-comparison}
	Suppose the scores in Equation~\eqref{eq:ate-variance} have finite second
	moments and $A,B>0$. Define
$\widetilde{\mathcal V}(\lambda,\omega) \triangleq n\operatorname{Var}\{\widehat\theta_{r_0}(\lambda,\omega)\}$ and $\sigma_0^2\triangleq\operatorname{Var}_R(Z_0)$. Here ``PPI\textsuperscript{++}-only'' means the member of our family that fixes $\lambda=0$ and optimally selects $\omega$. Then
\begin{align}
	\widetilde{\mathcal V}_{\mathrm{RCT}}
	&\triangleq\widetilde{\mathcal V}(0,0)=\sigma_0^2,
	\notag\\
	\widetilde{\mathcal V}_{\mathrm{PPI}^{++}}
	&\triangleq\widetilde{\mathcal V}(0,D/B)
	=\sigma_0^2-\frac{D^2}{B},
	\notag\\
	\widetilde{\mathcal V}_{\mathrm{joint}}
	&\triangleq\widetilde{\mathcal V}(-C/A,D/B)
	=\sigma_0^2-\frac{D^2}{B}-\frac{C^2}{A}.
	\label{eq:ate-three-way-variance}
\end{align}
Therefore $\widetilde{\mathcal V}_{\mathrm{joint}} \leq\widetilde{\mathcal V}_{\mathrm{PPI}^{++}} \leq\widetilde{\mathcal V}_{\mathrm{RCT}}$.
\end{theorem}

Thus, with population-oracle coefficients and the exact density ratio, the joint ATE estimator has variance no larger than either the PPI\textsuperscript{++}-only estimator or the RCT-only estimator.

\subsection{Estimable honest algorithm}

This subsection gives a finite-sample algorithm for estimating the two fusion coefficients.

We first define the feasible region over which $\lambda$ and $\omega$ are selected.

\begin{definition}[Coefficient rectangle and interval projection]
\label{def:interval-projection}
	Define the nonempty compact coefficient rectangle by
	\begin{align}
		\Theta_{\mathrm{rect}}
		&\triangleq
		[L_\lambda,U_\lambda]\times[L_\omega,U_\omega].
		\label{eq:ate-coefficient-rectangle}
	\end{align}
	For a closed interval $[L,U]$ with $L\leq U$, define
	\begin{align}
		\operatorname{Proj}_{[L,U]}(u)
		&\triangleq
		\min\{U,\max(L,u)\}.
		\label{eq:interval-projection}
	\end{align}
	This operator returns $u$ when it lies in the interval and otherwise returns
	the nearest endpoint.
\end{definition}
For positive $A,B$, the separated objective gives the coordinatewise projection
\begin{align*}
	\lambda_{\Theta_{\mathrm{rect}}}^\star
	&=\operatorname{Proj}_{[L_\lambda,U_\lambda]}(-C/A),
	&
	\omega_{\Theta_{\mathrm{rect}}}^\star
	&=\operatorname{Proj}_{[L_\omega,U_\omega]}(D/B).
\end{align*}
The population oracle weakly improves the variance objective whenever $(0,0)$ is feasible. Estimated coefficients have no universal finite-sample no-harm guarantee.

\subsubsection{Coefficient estimation and honest evaluation}

For $S\in\{R,O\}$, suppose $n_S^{\mathrm{tune}}\geq2$. Conditional on the fixed nuisance fits, $\widehat{\operatorname{Cov}}_S$ and $\widehat{\operatorname{Var}}_S$ denote the usual unbiased tuning-sample covariance and variance with denominator $n_S^{\mathrm{tune}}-1$. The factor $n/N$ below uses the evaluation-sample sizes whose variance is
optimized.

Let $\widehat r$ denote a density-ratio estimator fitted only on the nuisance samples. Under the common-covariate regime, set $\widehat r\triangleq 1$.

Using the tuning samples, compute the following four empirical analogues of $A$, $B$, $C$, and $D$ in Definition~\ref{def:ate-variance-objects}:
\begin{align}
\widehat A&\triangleq\widehat{\operatorname{Var}}_R(\Delta),
&
\widehat B&\triangleq\widehat{\operatorname{Var}}_R(\widehat{g})
+\frac nN\widehat{\operatorname{Var}}_O(\widehat r\widehat{g}),
\notag\\
\widehat C&\triangleq\widehat{\operatorname{Cov}}_R(Z_0,\Delta),
&
\widehat D&\triangleq\widehat{\operatorname{Cov}}_R(Z_0,\widehat{g}).
\label{eq:ate-empirical-moments}
\end{align}
These four moments define the separated empirical objective
\begin{align}
\widehat{\mathcal J}_{\mathrm{ATE}}(\lambda,\omega)
&\triangleq
2\widehat C\lambda-2\widehat D\omega
+\widehat A\lambda^2+\widehat B\omega^2.
\label{eq:ate-empirical-objective}
\end{align}

Before observing the tuning outcomes, prespecify the bounds defining the nonempty compact rectangle $\Theta_{\mathrm{rect}}$, require it to contain the origin, and choose positive thresholds $t_A$ and $t_B$.

Recall that $\widehat\mu_R=(\widehat\mu_{R,0},\widehat\mu_{R,1})$ and $\widehat\mu_O=(\widehat\mu_{O,0},\widehat\mu_{O,1})$ are outcome-regression predictors fitted on the RCT and OBS nuisance samples, respectively, and $\widehat{g}$ is their fitted OBS contrast from Equation~\eqref{eq:obs-effect-prediction}. Algorithm~\ref{alg:honest-ate} states the independent-sample implementation covered by the exact finite-sample results. The nuisance, tuning, and evaluation samples may be supplied directly or obtained by an honest prespecified split.

\begin{algorithm}[H]
\caption{ATE estimation with honest evaluation}
\label{alg:honest-ate}
\DontPrintSemicolon
\KwIn{Nuisance samples $D_R^{\mathrm{nuis}},D_O^{\mathrm{nuis}}$, tuning
samples $D_R^{\mathrm{tune}},D_O^{\mathrm{tune}}$, evaluation samples
$D_R^{\mathrm{eval}},D_O^{\mathrm{eval}}$, the rectangle
$\Theta_{\mathrm{rect}}$, and denominator thresholds.}
Fit $\widehat\mu_R$, $\widehat\mu_O$, $\widehat{g}$, and a density-ratio estimator $\widehat r$ on $D_R^{\mathrm{nuis}},D_O^{\mathrm{nuis}}$. The function $\widehat r$ estimates the exact density ratio $r_0$. When $\widehat r\neq r_0$, Equation~\eqref{eq:ate-ratio-drift} gives the resulting mean error. \;

Compute the four moments in Equation~\eqref{eq:ate-empirical-moments} on $D_R^{\mathrm{tune}},D_O^{\mathrm{tune}}$, and select the coefficients using the threshold-and-projection rule in Equation~\eqref{eq:ate-separated-plugin}.\;

Set $r=\widehat r$ in Equation~\eqref{eq:ate-estimator}, and evaluate it on $D_R^{\mathrm{eval}},D_O^{\mathrm{eval}}$. Set $\psi_R\triangleq Z_{\widehat\lambda}-\widehat\omega \widehat{g}$ and $\psi_O\triangleq\widehat\omega\widehat r\widehat{g}$. Compute $\widehat V\triangleq \widehat{\operatorname{Var}}_{R,\mathrm{eval}}(\psi_R)/n+\widehat{\operatorname{Var}}_{O,\mathrm{eval}}(\psi_O)/N$.\;
\KwOut{$\widehat\theta_{\widehat r}$ and $\widehat V$.}
\end{algorithm}

\paragraph{How Algorithm~\ref{alg:honest-ate} selects the coefficients.}

Because Equation~\eqref{eq:ate-empirical-objective} is separated in $\lambda$ and $\omega$, Algorithm~\ref{alg:honest-ate} selects the two coefficients coordinatewise and projects each one onto its prespecified interval:
\begin{align}
\widehat\lambda
&\triangleq
\begin{cases}
\operatorname{Proj}_{[L_\lambda,U_\lambda]}(-\widehat C/\widehat A),&\widehat A>t_A,\\
0,&\widehat A\leq t_A,
\end{cases}
&
\widehat\omega
&\triangleq
\begin{cases}
\operatorname{Proj}_{[L_\omega,U_\omega]}(\widehat D/\widehat B),&\widehat B>t_B,\\
0,&\widehat B\leq t_B.
\end{cases}
\label{eq:ate-separated-plugin}
\end{align}

\paragraph{Analysis.}

The four-moment rule in Equation~\eqref{eq:ate-separated-plugin} directly estimates the two population-oracle coordinates in Equation~\eqref{eq:axis-oracle}. The thresholds avoid division by nearly zero empirical curvature, and the projections enforce the prespecified feasible rectangle.



For Algorithm~\ref{alg:honest-ate}, set $r=\widehat r$ in the generic evaluation scores below.  For explicit evaluation notation, define
\begin{align*}
	\psi_{R,i}
	&\triangleq
	Z_{\widehat\lambda}(V_{R,i}^{\mathrm{eval}})-\widehat\omega \widehat{g}(X_{R,i}^{\mathrm{eval}}),
	&
	\psi_{O,j}
	&\triangleq
	\widehat\omega r(X_{O,j}^{\mathrm{eval}})\widehat{g}(X_{O,j}^{\mathrm{eval}}),
\end{align*}
Let $\widehat{\operatorname{Var}}_{R,\mathrm{eval}}$ and $\widehat{\operatorname{Var}}_{O,\mathrm{eval}}$ be the usual unbiased sample variances with denominators $n-1$ and $N-1$, respectively. Define the fixed-selected-coefficient variance estimator by
\begin{align}\label{eq:ate-evaluation-variance}
	\widehat V &\triangleq
	\frac{\widehat{\operatorname{Var}}_{R,\mathrm{eval}}(\psi_R)}{n}
	+
	\frac{\widehat{\operatorname{Var}}_{O,\mathrm{eval}}(\psi_O)}{N}.
\end{align}

\begin{proposition}[Exact variance estimation after honest selection]
\label{prop:ate-honest-variance-active}
	Let $\widehat\theta_r$ and $\widehat V$ be the estimator and variance estimator from Algorithm~\ref{alg:honest-ate}, specialized to $r=\widehat r=r_0$. Suppose Assumption~\ref{ass:honest-separation} holds, $n,N\geq2$, and $\psi_R\in L_2(P_R)$, $\psi_O\in L_2(P_O)$. Conditional on all fitted and selected objects,
	\begin{align*}
		\mathbb E(\widehat\theta_r)&=\theta,
		&
		\mathbb E(\widehat V)
		&=
		\operatorname{Var}\{\widehat\theta_r(\widehat\lambda,\widehat\omega)\}.
	\end{align*}
	When the corresponding unconditional moments exist, the same identities also hold unconditionally because the conditional mean is the nonrandom target $\theta$.
\end{proposition}


\begin{theorem}[Normal approximation after honest evaluation]
\label{thm:ate-honest-clt-active}

Let $\widehat\theta_r$ and $\widehat V$ be the estimator and variance
estimator from Algorithm~\ref{alg:honest-ate}, specialized to
$r=\widehat r=r_0$. Suppose Assumption~\ref{ass:honest-separation} holds, $n,N\to\infty$, $n/N\to\kappa\in[0,\infty)$, and their centered scores
have uniformly bounded conditional $(2+\delta)$-moments for some $\delta>0$.

Define the conditional evaluation variance
$V_n\triangleq\operatorname{Var}\{\widehat\theta_r\mid\text{fitted and selected objects}\}$ and assume
$nV_n\to_p\sigma^2\in(0,\infty)$ and
$\widehat V/V_n\to_p 1$. Then
\begin{align*}
\frac{\widehat\theta_r-\theta}{\sqrt{\widehat V}}
&\rightsquigarrow N(0,1),
\end{align*}
and
\begin{align*}
\widehat\theta_r
&\mathbin{\pm}z_{1-\alpha/2}\sqrt{\widehat V}
\end{align*}
has asymptotic coverage $1-\alpha$, where $z_{1-\alpha/2}$ is the standard
normal quantile. Variance unbiasedness alone does not imply this coverage.
\end{theorem}

Proposition~\ref{prop:ate-honest-variance-active} and
Theorem~\ref{thm:ate-honest-clt-active} are the exact-ratio benchmarks. The next
subsection develops feasible inference when Algorithm~\ref{alg:honest-ate} uses
an estimated ratio $\widehat r$ that need not equal $r_0$.

\subsubsection{Estimating the density ratio}
\label{sec:ratio-estimation}

This subsection explains how to estimate the target density ratio $r_0$, replacing the fixed candidate $r$ above by a fitted ratio $\widehat r$.

\begin{proposition}[Transport characterization]
\label{prop:ratio-characterizations}
	Under either Assumption~\ref{ass:common-marginal} or Assumption~\ref{ass:exact-transport}, every $h\in L_1(P_R^X)$ satisfies
	\begin{align}
		\mathbb E_O\{r_0(X)h(X)\} &=\mathbb E_R\{h(X)\}.
		\label{eq:ratio-transport-characterization}
	\end{align}
	Conversely, if $\widetilde r\geq0$, $\mathbb E_O(\widetilde r)=1$, and Equation~\eqref{eq:ratio-transport-characterization} holds with $\widetilde r$ in place of $r_0$ for every bounded measurable $h$, then $\widetilde r=r_0$ $P_O^X$-almost surely.
\end{proposition}

\Needspace{12\baselineskip}
\begin{proposition}[Riesz-risk characterization]
\label{prop:ratio-riesz-characterization}
If $r_0\in L_2(P_O^X)$, then
\begin{align}\label{eq:ratio-riesz-population}
	r_0 &\in\operatorname*{arg\,min}_{a\in L_2(P_O^X)}\mathcal \mathbb E_O\{a(X)^2\}-2\mathbb E_R\{a(X)\},
\end{align}
The minimizer is unique up to $P_O^X$-almost-sure equality.
\end{proposition}

Eq.~\eqref{eq:ratio-riesz-population} motivates the following learner.

\begin{definition}[Empirical Riesz ratio learner]
\label{def:empirical-riesz-ratio}
	For $S\in\{R,O\}$, let $\mathbb P_S^{\mathrm{nuis}}$ denote the
	empirical average over $D_S^{\mathrm{nuis}}$. For a prespecified function class $\mathcal A$, penalty $\operatorname{Pen}$, and $\rho\geq0$, use a prespecified deterministic rule to break ties among minimizers. Whenever the displayed argmin set is nonempty, choose
	\begin{align}
		\widehat r_{\mathrm{RZ}}
		&\in
		\operatorname*{arg\,min}_{a\in\mathcal A}
		\left[
		\mathbb P_O^{\mathrm{nuis}}(a^2)-2\mathbb P_R^{\mathrm{nuis}}(a)
		+\rho\operatorname{Pen}(a)
		\right].
		\label{eq:ratio-riesz-empirical}
	\end{align}
	The classifier and Riesz routes are alternative base learners; we do not average them.
\end{definition}

Equation~\eqref{eq:ratio-riesz-empirical} is Riesz-representer learning of the type used in Auto-DML \citep{chernozhukov2021riesz,chernozhukov2022autodml}.

The next lemma specializes Definition~\ref{def:empirical-riesz-ratio} to a linear-ridge class and gives explicit conditions for a unique solution.
\begin{lemma}[Linear Riesz solution]
\label{lem:linear-riesz-ratio}
Let $a_\gamma(x)\triangleq\phi(x)^\top\gamma$, where $\phi:\mathcal X\to\mathbb R^p$. Let $\operatorname{Pen}(a_\gamma)\triangleq\|\gamma\|_2^2$.
The empirical Gram matrix, namely the average of the feature outer products over the OBS nuisance sample, is $\mathbb P_O^{\mathrm{nuis}}\{\phi(X)\phi(X)^\top\}$. If $\rho>0$, or if this matrix is nonsingular when $\rho=0$, then
\begin{align}
	\widehat\gamma_\rho
	&\triangleq
	\left[
	\mathbb P_O^{\mathrm{nuis}}\{\phi(X)\phi(X)^\top\}+\rho I_p
	\right]^{-1}
	\mathbb P_R^{\mathrm{nuis}}\{\phi(X)\}.
	\label{eq:ratio-riesz-linear}
\end{align}
\end{lemma}

The classifier learns the ratio indirectly through source prediction.
\begin{definition}[Classifier density-ratio estimator]
\label{def:classifier-density-ratio}
Pool the nuisance-sample covariates and label their source by $S=1$ for RCT and $S=0$ for OBS. Define
\begin{align*}
	\pi &\triangleq P(S=1),
	&
	s_0(x) &\triangleq P(S=1\mid X=x).
\end{align*}
When $0<\pi<1$, Bayes' rule gives
\begin{align}\label{eq:ratio-classifier-bayes}
	r_0(x) &= \frac{1-\pi}{\pi}\frac{s_0(x)}{1-s_0(x)}.
\end{align}
For a fitted domain probability $\widehat s$, let $\widehat\pi$ denote the effective RCT proportion in classifier training and define
\begin{align}\label{eq:ratio-classifier-estimator}
	\widehat r_{\mathrm{CLS}}(x) &\triangleq
	\frac{1-\widehat\pi}{\widehat\pi}
	\frac{\widehat s(x)}{1-\widehat s(x)}.
\end{align}
\end{definition}

For an unweighted pooled classifier,
$\widehat\pi\triangleq
n_R^{\mathrm{nuis}}/(n_R^{\mathrm{nuis}}+n_O^{\mathrm{nuis}})$. Class
weighting or resampling changes the effective training prior, so the prior-odds
factor must be corrected accordingly. The odds require
$0<\widehat s(x)<1$ where the ratio is used. Probability clipping can
stabilize the estimate, but it changes the exact density-ratio target.

The Riesz learner in Definition~\ref{def:empirical-riesz-ratio} and the classifier in Definition~\ref{def:classifier-density-ratio} are the two primary ratio-estimation routes. After either route has been fitted, one may optionally calibrate its positive ratio to selected ATE features.

\begin{definition}[Calibrated automatic-balancing ratio]
\label{def:classifier-calibrated-balance}
Let $\widehat r_{\mathrm{base}}>0$ come from either the classifier or Riesz
route. Let $\phi:\mathcal X\to\mathbb R^p$ be prespecified with no intercept.
Whenever the displayed argmin set is nonempty, choose a deterministically
tie-broken minimizer
\begin{align}
	\widehat\xi
	&\in\operatorname*{arg\,min}_{\xi\in\mathbb R^p}
	\left[
	\log\mathbb P_O^{\mathrm{nuis}}
	\{\widehat r_{\mathrm{base}}(X)e^{\xi^\top\phi(X)}\}
	-\xi^\top\mathbb P_R^{\mathrm{nuis}}\{\phi(X)\}
	\right],
	\label{eq:ratio-calibration-objective}
	\\
	\widehat r_{\mathrm{BAL}}(x)
	&\triangleq
	\frac{\widehat r_{\mathrm{base}}(x)e^{\widehat\xi^\top\phi(x)}}
	{\mathbb P_O^{\mathrm{nuis}}
	\{\widehat r_{\mathrm{base}}(X)e^{\widehat\xi^\top\phi(X)}\}}.
	\label{eq:ratio-calibrated-estimator}
\end{align}
\end{definition}

Normalization already balances the constant, so an intercept would be
nonidentified. The minimal ATE feature is
$\phi_{\mathrm{ATE}}(X)\triangleq\widehat{g}(X)$; a richer finite map can add
$\widehat{g}^2$, fitted regressions, or a small spline or frozen-neural basis.
This exponential tilt follows the calibration principle behind entropy
balancing \citep{hainmueller2012entropy}.

\begin{proposition}[Automatic balance]
\label{prop:automatic-ratio-balance}
If the objective in Definition~\ref{def:classifier-calibrated-balance} has an
interior minimizer, then $\widehat r_{\mathrm{BAL}}>0$,
$\mathbb P_O^{\mathrm{nuis}}(\widehat r_{\mathrm{BAL}})=1$, and
\begin{align}
\mathbb P_O^{\mathrm{nuis}}
\{\widehat r_{\mathrm{BAL}}(X)\phi(X)\}
&=\mathbb P_R^{\mathrm{nuis}}\{\phi(X)\}.
\label{eq:ratio-calibration-balance}
\end{align}
\end{proposition}

These are $p$ finite moment equations, not pointwise balance. They are analogous to covariate-balancing propensity scores, which concern treatment rather than source assignment \citep{imai2014covariate}. Exact feasibility requires the RCT feature mean to lie in the relative interior of the convex hull of the positively weighted OBS feature vectors; otherwise a prespecified approximate-balance rule is required.

We now quantify how ratio error propagates to coefficient selection and the final ATE estimator.
Recall the tuning-sample moments $\widehat A,\widehat B,\widehat C,\widehat D$ in Equation~\eqref{eq:ate-empirical-moments}, the selection rule with thresholds $t_A,t_B$ in Equation~\eqref{eq:ate-separated-plugin}, the variance estimator $\widehat V$ in Equation~\eqref{eq:ate-evaluation-variance}, and the conditional evaluation variance $V_n$ in Theorem~\ref{thm:ate-honest-clt-active}. Below, all of them are computed with the estimated ratio $\widehat r$ in place of $r_0$.
\begin{proposition}[How ratio error changes coefficient selection]
\label{lem:ratio-error-coefficients}
Define the ratio-weighted prediction error by
\begin{align}\label{eq:ratio-error-score}
	\delta_{\widehat r}(X) &\triangleq\{\widehat r(X)-r_0(X)\}\widehat g(X).
\end{align}
For a ratio $r\in\{\widehat r,r_0\}$, let $\widehat B_r$ denote the tuning-sample moment $\widehat B$ of Equation~\eqref{eq:ate-empirical-moments} computed with $r$:
\begin{align}\label{eq:ratio-dependent-B}
	\widehat B_{r} &\triangleq \widehat{\operatorname{Var}}_R(\widehat g) +\frac nN\widehat{\operatorname{Var}}_O(r\widehat g).
\end{align}
Thus $\widehat B_{\widehat r}$ is the denominator that Algorithm~\ref{alg:honest-ate} actually uses for $\widehat\omega$ in Equation~\eqref{eq:ate-separated-plugin}, and $\widehat B_{r_0}$ is the denominator it would use if $r_0$ were known. The numerator $\widehat D=\widehat{\operatorname{Cov}}_R(Z_0,\widehat g)$ uses the RCT tuning sample only and does not depend on $r$. Then
\begin{align}\label{eq:ratio-B-error}
	\widehat B_{\widehat r}-\widehat B_{r_0} &=
	\frac nN\left[
	2\widehat{\operatorname{Cov}}_O(r_0\widehat g,\delta_{\widehat r})
	+\widehat{\operatorname{Var}}_O(\delta_{\widehat r})
	\right].
\end{align}
Hence ratio error enters $\widehat\omega=\widehat D/\widehat B$ only through this denominator and the threshold $t_B$ in Equation~\eqref{eq:ate-separated-plugin}. It leaves $\widehat\lambda$ unchanged because the separated rule computes $\widehat\lambda$ only from $\widehat A$ and $\widehat C$.
\end{proposition}
The next proposition isolates its additional, direct effect on the final ATE estimator.

\begin{proposition}[Exact estimated-ratio decomposition]
\label{prop:estimated-ratio-decomposition}
	Let $(\widehat\lambda,\widehat\omega)$ be selected without the evaluation
	observations. Define the scalar population drift by
	\begin{align*}
	b_{\widehat r}&\triangleq\widehat\omega P_O\delta_{\widehat r}.
	\end{align*}
	Conditional on the fitted objects and selected coefficients,
	\begin{align}
		\widehat\theta_{\widehat r}(\widehat\lambda,\widehat\omega)-\theta
		&=
		\widehat\theta_{r_0}(\widehat\lambda,\widehat\omega)-\theta
		+b_{\widehat r}
		+\widehat\omega(\mathbb P_{O,N}-P_O)\delta_{\widehat r}.
		\label{eq:estimated-ratio-mean-fluctuation}
	\end{align}
\end{proposition}
Equation~\eqref{eq:ratio-B-error} describes the coefficient-selection path, whereas Equation~\eqref{eq:estimated-ratio-mean-fluctuation} separates the direct ATE error into population drift and evaluation fluctuation.






\begin{proposition}[Conditional mean and variance with an estimated ratio]
\label{cor:estimated-ratio-mean-preservation}
Run Algorithm~\ref{alg:honest-ate} with the estimated ratio $\widehat r$, so that $\widehat\theta_{\widehat r}$ and $\widehat V$ in Equation~\eqref{eq:ate-evaluation-variance} are computed with $\widehat r$. Suppose Assumption~\ref{ass:honest-separation} holds, $n,N\geq2$, and $\psi_R\in L_2(P_R)$, $\psi_O\in L_2(P_O)$. Suppose also that $\widehat{g}\in L_1(P_R)$ and
$r_0\widehat{g},\widehat r\widehat{g}\in L_1(P_O)$. With
$b_{\widehat r}$ defined in
Proposition~\ref{prop:estimated-ratio-decomposition}, let $V_n$ be the conditional evaluation variance of Theorem~\ref{thm:ate-honest-clt-active} computed with $\widehat r$:
\begin{align}
V_n
&\triangleq
\operatorname{Var}\{\widehat\theta_{\widehat r}\mid
\text{fitted and selected objects}\}.
\label{eq:estimated-ratio-conditional-variance}
\end{align}
Then
\begin{align}
\mathbb E\{\widehat\theta_{\widehat r}\mid
\text{fitted and selected objects}\}
&=\theta+b_{\widehat r},
\\
\mathbb E\{\widehat V\mid
\text{fitted and selected objects}\}
&=V_n.
\label{eq:estimated-ratio-conditional-identities}
\end{align}
The conditional mean equals $\theta$ exactly when $b_{\widehat r}=0$. The
balance condition
\begin{align}
P_O\delta_{\widehat r}=0
\quad\Longleftrightarrow\quad
\mathbb E_O\{\widehat r(X)\widehat{g}(X)\}
&=\mathbb E_R\{\widehat{g}(X)\}
\label{eq:estimated-ratio-mean-preservation}
\end{align}
is sufficient, and $\widehat\omega=0$ is another sufficient case.
\end{proposition}

When the corresponding unconditional moments exist,
$\mathbb E(\widehat\theta_{\widehat r})=\theta+\mathbb E(b_{\widehat r})$ and
$\mathbb E(\widehat V)=\mathbb E(V_n)$. In general,
the latter is not the unconditional variance of the estimator, which also
contains the between-fit variation of $b_{\widehat r}$.





\begin{theorem}[Feasible normal approximation with an estimated ratio]
\label{thm:estimated-ratio-feasible-clt}
Let $\widehat\theta_{\widehat r}$, $\widehat V$, and $V_n$ be as in Proposition~\ref{cor:estimated-ratio-mean-preservation}. Suppose Assumption~\ref{ass:honest-separation} holds, $n,N\to\infty$, $n/N\to\kappa\in[0,\infty)$, and,
for some $\delta>0$, the centered evaluation scores have conditional
$(2+\delta)$-moments that are uniformly bounded in probability. If
\begin{align}
nV_n&\to_p\sigma^2\in(0,\infty),
&
\frac{b_{\widehat r}}{\sqrt{V_n}}&=o_p(1),
\label{eq:estimated-ratio-mean-rate}
\end{align}
then
\begin{align}
\frac{\widehat V}{V_n}&\to_p1,
&
\frac{\widehat\theta_{\widehat r}-\theta}
{\sqrt{\widehat V}}&\rightsquigarrow N(0,1).
\label{eq:estimated-ratio-feasible-clt}
\end{align}
Consequently, the interval
\begin{align}
\widehat\theta_{\widehat r}
&\mathbin{\pm}z_{1-\alpha/2}\sqrt{\widehat V}
\label{eq:estimated-ratio-confidence-interval}
\end{align}
has asymptotic coverage $1-\alpha$, where $z_{1-\alpha/2}$ is the standard
normal quantile.
\end{theorem}

Theorem~\ref{thm:estimated-ratio-feasible-clt} permits
$\widehat r\neq r_0$, but it requires the induced conditional mean drift to be
negligible relative to the standard error. If that drift is not negligible, the
present interval needs further justification. Accommodating first-stage
variation requires an appropriate asymptotic expansion and variance analysis,
which may use influence functions, direct bias control, or a different
orthogonal construction.

\begin{theoremcorollary}[Inheritance of the exact-ratio oracle ordering]
\label{cor:estimated-ratio-inheritance}
Index the three estimators of Theorem~\ref{thm:ate-three-way-comparison}, built from the same fitted objects $Z_0,\Delta,\widehat g,\widehat r$, by $k\in\{\mathrm{RCT},\mathrm{PPI}^{++},\mathrm{joint}\}$. Their population-oracle coefficients are
\begin{align*}
(\lambda^\star_{\mathrm{RCT}},\omega^\star_{\mathrm{RCT}})&\triangleq(0,0),
&
(\lambda^\star_{\mathrm{PPI}^{++}},\omega^\star_{\mathrm{PPI}^{++}})&\triangleq(0,D/B),
\\
(\lambda^\star_{\mathrm{joint}},\omega^\star_{\mathrm{joint}})&\triangleq(-C/A,D/B),
\end{align*}
and their feasible coefficients $(\widehat\lambda_k,\widehat\omega_k)$ are the entries of Equation~\eqref{eq:ate-separated-plugin}, with $\widehat\lambda_k\triangleq0$ for $k\neq\mathrm{joint}$ and $\widehat\omega_{\mathrm{RCT}}\triangleq0$. Write $\widehat\theta_{\widehat r,k}\triangleq\widehat\theta_{\widehat r}(\widehat\lambda_k,\widehat\omega_k)$, and let $\sigma_k^2$ denote the limit $\sigma^2$ in Equation~\eqref{eq:estimated-ratio-mean-rate} for estimator $k$. Suppose that, for each $k$, the conditions of Theorem~\ref{thm:estimated-ratio-feasible-clt} hold, $(\widehat A,\widehat B,\widehat C,\widehat D)-(A,B,C,D)=o_p(1)$, $(\widehat\lambda_k,\widehat\omega_k)-(\lambda_k^\star,\omega_k^\star)=o_p(1)$,
\begin{align}
\operatorname{Var}_R(\Delta)&=O_p(1),
&
\operatorname{Var}_R(\widehat g)+\frac nN\operatorname{Var}_O(r_0\widehat g)&=O_p(1),
\notag\\
\mathbb P(\widehat A\leq t_A)+\mathbb P(\widehat B\leq t_B)&\to0,
\label{eq:estimated-ratio-stability}
\end{align}
and
\begin{align}
|\widehat\omega_k|\sqrt{\frac nN}
\left\|\delta_{\widehat r}-P_O\delta_{\widehat r}\right\|_{L_2(P_O)}
&=o_p(1).
\label{eq:estimated-ratio-fluctuation-rate}
\end{align}
Then, for each $k$,
\begin{align}
\sqrt n\left\{\widehat\theta_{\widehat r,k}-\widehat\theta_{r_0}(\lambda_k^\star,\omega_k^\star)\right\}
&=o_p(1),
\label{eq:estimated-ratio-first-order-equivalence}
\end{align}
so $\sigma_k^2$ equals the limit of $\widetilde{\mathcal V}_k$ in Equation~\eqref{eq:ate-three-way-variance} and
\begin{align}
\sigma^2_{\mathrm{joint}}&\leq\sigma^2_{\mathrm{PPI}^{++}}\leq\sigma^2_{\mathrm{RCT}}.
\label{eq:estimated-ratio-inherited-ordering}
\end{align}
\end{theoremcorollary}
This is not a universal
finite-sample dominance statement.
