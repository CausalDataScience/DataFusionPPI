\section{Background and Problem Setting}
\label{sec:setup}

\subsection{RCT--OBS problem setting and notation}
\label{sec:rct-obs-setting}

Let $\mathcal K\triangleq\{\mathrm{eval},\mathrm{nuis},\mathrm{tune}\}$ index the three statistical roles assigned to the samples. For each source $S\in\{R,O\}$, where $R$ and $O$ denote the randomized and observational sources, respectively, and each role $k\in\mathcal K$, define
\begin{align}
	D_S^k &\triangleq
	\{V_{S,i}^k=(X_{S,i}^k,A_{S,i}^k,Y_{S,i}^k):i=1,\ldots,n_S^k\}
	\sim P_S^{\otimes n_S^k},
	\label{eq:role-indexed-samples}
\end{align}
as the sample from source $S$ reserved for role $k$. Here $X$ takes values in the covariate space $\mathcal X$, $A\in\{0,1\}$ is treatment, and $Y$ is the observed outcome. For each $S\in\{R,O\}$, the datasets $D_S^{\mathrm{eval}}$, $D_S^{\mathrm{nuis}}$, and $D_S^{\mathrm{tune}}$ are used for final evaluation, nuisance and density-ratio fitting, and coefficient or candidate tuning, respectively.

We impose the following independence condition across the six role-specific samples.
\begin{assumption}[Honest sample separation]
\label{ass:honest-separation}
	The six datasets $\{D_S^k:S\in\{R,O\},\ k\in\mathcal K\}$ are mutually independent.
\end{assumption}

Set $n\triangleq n_R^{\mathrm{eval}}$ and $N\triangleq n_O^{\mathrm{eval}}$, and define $\mathbb P_{R,n}h\triangleq n^{-1}\sum_i h(V_{R,i}^{\mathrm{eval}})$ and $\mathbb P_{O,N}h\triangleq N^{-1}\sum_j h(V_{O,j}^{\mathrm{eval}})$. Every displayed moment is finite; square-integrability and stronger moment conditions are stated locally when variance estimation or normal-approximation inference requires them.

\Needspace{6\baselineskip}
Let $Y(1)$ and $Y(0)$ denote potential outcomes. The RCT-population conditional and average treatment effects are
\begin{align}
	\tau(x) &\triangleq\mathbb E_R\{Y(1)-Y(0)\mid X=x\},
	\\ 
	\theta &\triangleq\mathbb E_R\{\tau(X)\}.
\end{align}

For causal identification in the trial population, we impose the following assumptions under $P_R$.
\begin{assumption}[Trial identification]
\label{ass:trial}
	The following assumptions hold: 
	\begin{enumerate}
		\item $Y=Y(A)$ almost surely;
		\item $\{Y(0),Y(1)\}\perp A\mid X$ under the interventional distribution $P_R$; 
		\item The trial propensity $e(x)\triangleq P_R(A=1\mid X=x)$ is known and satisfies the positivity: $\epsilon<e(X)<1-\epsilon$ almost surely for a constant $\epsilon>0$.
	\end{enumerate}
\end{assumption}
We study both the common-covariate regime $P_R^X=P_O^X$, formalized in Assumption~\ref{ass:common-marginal}, and the more general covariate-shift regime $P_R^X\neq P_O^X$, formalized in Assumption~\ref{ass:exact-transport}.

\begin{definition}[Target-to-source density ratio]
\label{def:true-density-ratio}
	Whenever $P_R^X\ll P_O^X$, define the target-to-source density ratio by
	\begin{align*}
		r_0(x) &\triangleq \frac{dP_R^X}{dP_O^X}(x).
	\end{align*}
	It is the Radon--Nikodym derivative that transports an OBS covariate expectation
	to the RCT covariate population.
\end{definition}

\enlargethispage{\baselineskip}
\Needspace{5\baselineskip}
\begin{assumption}[Common covariate marginal]
	\label{ass:common-marginal}
	\label{ass:covariate}
	The trial and observational covariates have the same marginal law, $P_R^X=P_O^X$. This assumption does not equate the treatment or outcome mechanisms across sources, and the observational treatment contrast may remain confounded. In this regime, Definition~\ref{def:true-density-ratio} gives $r_0(x)=1$ for $P_O^X$-almost every $x$.
\end{assumption}

\begin{assumption}[Covariate shift and support overlap]
\label{ass:exact-transport}
	Under covariate shift, $P_R^X\ll P_O^X$, so the RCT covariate population has
	no support outside the OBS covariate population. Definition~\ref{def:true-density-ratio}
	therefore defines $r_0$. This assumption establishes existence of the ratio;
	it does not assert that $r_0$ is known or exactly estimated.
\end{assumption}

Assumptions~\ref{ass:common-marginal} and \ref{ass:exact-transport} describe alternative covariate regimes, not cumulative restrictions. Under Assumption~\ref{ass:common-marginal}, the exact density ratio is $r_0=1$, so no transport weighting is needed. When the covariate marginals differ, Assumption~\ref{ass:exact-transport} is needed to convert OBS covariate expectations into expectations under the RCT population.

For $a\in\{0,1\}$, define the RCT outcome-regression target $\mu_{R,a}:\mathcal X\to\mathbb R$ by
\begin{align*}
	\mu_{R,a}(x) &\triangleq \mathbb E_R[Y\mid X=x,A=a].
\end{align*}
For each $a\in\{0,1\}$, let $\widehat\mu_{R,a}:\mathcal X\to\mathbb R$ denote a fixed pre-evaluation predictor of $\mu_{R,a}$ trained only on $D_R^{\mathrm{nuis}}$. Define
\begin{align}
	\widehat\mu_R &\triangleq (\widehat\mu_{R,0},\widehat\mu_{R,1}).
	\label{eq:rct-outcome-regression}
\end{align}

We define the following AIPW score.
\begin{definition}[RCT-valid AIPW score]
\label{def:rct-aipw-score}
	Let $q\triangleq(q_0,q_1)$ be a fixed pair of measurable functions from
	$\mathcal X$ to $\mathbb R$. Define
	\begin{align}
		\varphi(V;q) &\triangleq q_1(X)-q_0(X) +\frac{A}{e(X)}\{Y-q_1(X)\} -\frac{1-A}{1-e(X)}\{Y-q_0(X)\}.
		\label{eq:aipw-score}
	\end{align}
\end{definition}

This score gives a valid causal estimate: 
\Needspace{7\baselineskip}
\begin{proposition}[RCT-validity for an arbitrary outcome regression]
	\label{prop:pseudo-validity-active}
	\label{prop:pseudo-validity}
	Suppose Assumption~\ref{ass:trial} holds. Let $q$ be a measurable function pair fixed independently of the RCT evaluation sample, and suppose $\varphi(V;q)$ is integrable. Then
	\begin{align}
		\mathbb E_R\{\varphi(V;q)\mid X\} &=\tau(X).
		\label{eq:pseudo-valid}
	\end{align}
\end{proposition}

\begin{definition}[RCT-only AIPW estimator]
\label{def:rct-only-aipw}
	Define the RCT-only AIPW estimator on the evaluation sample by
	\begin{align}
		\widehat\theta_{\mathrm{AIPW},R}
		&\triangleq
		\mathbb P_{R,n}\varphi(V;\widehat\mu_R).
		\label{eq:rct-only-aipw}
	\end{align}
\end{definition}

\begin{corollary}[RCT-only AIPW recovery]
\label{cor:rct-only-aipw}
	Suppose Assumptions~\ref{ass:trial} and \ref{ass:honest-separation} hold and
	$\varphi(V;\widehat\mu_R)$ is integrable. Conditional on the nuisance data
	used to fit $\widehat\mu_R$,
	$\mathbb E\{\widehat\theta_{\mathrm{AIPW},R}\mid D_R^{\mathrm{nuis}}\}=\theta$.
\end{corollary}

For $a\in\{0,1\}$, define the OBS outcome-regression target $\mu_{O,a}:\mathcal X\to\mathbb R$ by
\begin{align*}
	\mu_{O,a}(x) &\triangleq \mathbb E_O[Y\mid X=x,A=a].
\end{align*}
For each $a\in\{0,1\}$, let $\widehat\mu_{O,a}:\mathcal X\to\mathbb R$ denote a fixed pre-evaluation predictor of $\mu_{O,a}$ trained only on $D_O^{\mathrm{nuis}}$. Define
\begin{align}
	\widehat\mu_O &\triangleq (\widehat\mu_{O,0},\widehat\mu_{O,1}).
	\label{eq:obs-outcome-regression}
\end{align}

The RCT-validity result also permits the RCT and OBS outcome regressions to be
combined without changing the conditional causal target.
\begin{definition}[Fusion score]
\label{def:fusion-score}
	For $\lambda\in\mathbb R$, define
	\begin{align}
		\widehat{\mu}_{\lambda} &\triangleq (1-\lambda)\widehat\mu_R+\lambda\widehat\mu_O,
	\end{align}
	and the fused score as
	\begin{align}
		Z_{\lambda} &\triangleq \varphi(V;\widehat\mu_\lambda).
		\label{eq:fusion-score}
	\end{align}
\end{definition}
From $Z_0 = \varphi(V;\widehat\mu_R)$, $Z_1 = \varphi(V;\widehat\mu_O)$ and $\Delta \triangleq Z_1-Z_0$, we can rewrite the fused score as
\begin{align}
	Z_{\lambda} = Z_0 + \lambda \Delta.
\end{align}




\subsection{Existing data-fusion estimators}
\label{sec:existing-fusion}

Data integration first asks whether information from different regimes or populations identifies a common target. Transport and general data-fusion frameworks make the source and target populations explicit \citep{bareinboim2016fusion,shi2022integration,colnet2020causal}. Our target is already identified by the RCT; transport is used to align fixed OBS covariate functions with the RCT population.

Robust borrowing methods instead trade potential efficiency gains against observational bias. They include shrinkage, compatibility testing, and adaptive weighting that returns toward the randomized estimate when the sources disagree \citep{rosenman2020shrinkage,cheng2021adaptive}. These methods clarify why observational information can help without implying a universal gain.

For heterogeneous effects, direct multi-source learners combine weighted pseudo-outcomes, while experimental-grounding and RCT--OBS procedures use the trial to constrain or diagnose observational HTE estimates \citep{li2022directfusion,kallus2018removing}. The elastic procedure of \citet{yang2023elastic} tests compatibility before pooling real-world data for prespecified effect modifiers. The confounding-function approach of \citet{yang2025confounding} jointly models an HTE function and an observational confounding function under explicit identification conditions.

Our construction differs at the fusion object. The coefficient $\lambda$ changes the outcome regression inside an RCT-valid score, whereas the second coefficient scales a transported mean-zero control variate built from a fitted OBS treatment-effect prediction. This PPI-style use of prediction does not combine two causal effect estimates and does not require the OBS contrast to equal the CATE.
